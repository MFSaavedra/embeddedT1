1"""@file accel_plots.py
2@brief Three sliding-window plots, one per axis (spec 2.2.3) - the highest-weighted GUI item.
3
4Design (see TODO.md PY4):
5- one numpy ring buffer of (t, value) per axis, sized for the longest window at 1000 Hz;
6- push() is called from the serial signal for every ACC frame and must only append;
7- a single QTimer repaints all three curves at ~30 fps - never redraw per sample;
8- the x axis is in seconds, so a change of fs shows up as denser/sparser points; the
9 y range follows the configured amplitude (+-A), so a change of A is obvious at once;
10- each plot title shows the *measured* incoming rate, averaged over RATE_WINDOW_S so it
11 reads steadily instead of flickering between quantised per-tick values; it doubles as a
12 link-health check (a rate below the configured fs means samples are being dropped).
13"""

94 def push(self, axis: str, t0_ms: int, fs_hz: int, values: Sequence[float]) -> None:
95 """@brief Append one batch of samples for @p axis (called for every ACC frame).
96
97 Timestamps are reconstructed as t0 + i / fs. Writes wrap around the ring buffer;
98 nothing is drawn here.
99
100 @param axis "X", "Y" or "Z".
101 @param t0_ms Timestamp of values[0] in ms, as sent by the firmware.
102 @param fs_hz Sample rate of the batch in Hz (consecutive samples are 1/fs apart).
103 @param values Sample values in g, oldest first; an empty batch is ignored.
104 """
105 n = len(values)
106 if n == 0:
107 return
108 t = t0_ms / 1000.0 + np.arange(n) / fs_hz
109 v = np.asarray(values, dtype=float)
110
111 t_buf, v_buf, idx, cap = self._t[axis], self._v[axis], self._idx[axis], self._cap
112 end = idx + n
113 if end <= cap:
114 t_buf[idx:end] = t
115 v_buf[idx:end] = v
116 else:
117 first = cap - idx
118 t_buf[idx:cap], v_buf[idx:cap] = t[:first], v[:first]
119 t_buf[:end - cap], v_buf[:end - cap] = t[first:], v[first:]
120
121 self._idx[axis] = end % cap
122 self._filled[axis] = min(cap, self._filled[axis] + n)
123 self._arrivals[axis] += n

147 def _redraw(self) -> None:
148 """@brief Timer slot: update the measured rate in each title and repaint the curves.
149
150 Unrolls the ring buffer in chronological order, keeps only the last window_s
151 seconds and scrolls the x range so the newest sample sits at the right edge.
152 The measured rate is recomputed once every RATE_WINDOW_S from the samples that
153 arrived in that window (arrivals / elapsed), not per redraw tick.
154 """
155 elapsed = time.monotonic() - self._rate_t0
156 update_rate = elapsed >= RATE_WINDOW_S
157 for axis in AXES:
158 filled = self._filled[axis]
159 if update_rate:
160 self._rate[axis] = self._arrivals[axis] / elapsed
161 self._arrivals[axis] = 0
162 self._plots[axis].setTitle(f"Eje {axis} — {self._rate[axis]:.0f} muestras/s")
163 if filled == 0:
164 continue
165
166 idx, cap = self._idx[axis], self._cap
167 if filled < cap:
168 t, v = self._t[axis][:filled], self._v[axis][:filled]
169 else:
170 t = np.concatenate((self._t[axis][idx:], self._t[axis][:idx]))
171 v = np.concatenate((self._v[axis][idx:], self._v[axis][:idx]))
172
173 t_last = t[-1]
174 mask = t >= t_last - self.window_s
175 self._curves[axis].setData(t[mask], v[mask])
176 self._plots[axis].setXRange(t_last - self.window_s, t_last)
177 if update_rate:
178 self._rate_t0 += elapsed
