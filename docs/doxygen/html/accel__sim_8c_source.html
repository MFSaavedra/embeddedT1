62float accel_sim_eval(wave_func_t func, float amp_g, float t_s)
63{
64 switch (func) {
65 case WAVE_SINE:
66 /* a1(t) = A * sin(2*pi*f*t) */
67 return amp_g * sinf(2.0f * (float)M_PI * ACCEL_F_HZ * t_s);
68 case WAVE_AM:
69 /* a2(t) = A * cos(2*pi*f1*t) * sin(2*pi*f2*t) */
70 return amp_g * cosf(2.0f * (float)M_PI * ACCEL_F1_HZ * t_s)
71 * sinf(2.0f * (float)M_PI * ACCEL_F2_HZ * t_s);
72 case WAVE_MULTI:
73 /* a3(t) = (2A/2) * [sin(2*pi*f*t) + cos(4*pi*f*t)] -- literal spec factor, peak = 2A */
74 return (2.0f * amp_g / 2.0f)
75 * (sinf(2.0f * (float)M_PI * ACCEL_F_HZ * t_s)
76 + cosf(4.0f * (float)M_PI * ACCEL_F_HZ * t_s));
77 default:
78 return 0.0f;
79 }
80}

94static void on_tick(void *arg)
95{
96 (void)arg;
97 if (!s_running) {
98 return;
99 }
100 s_tick++;
101
102 for (int a = 0; a < AXIS_COUNT; a++) {
103 axis_cfg_t cfg = accel_sim_get_axis((axis_t)a);
104 uint32_t divider = ACCEL_TICK_HZ / cfg.fs_hz; /* 1, 2, 5, 10 or 20 */
105 if (s_tick % divider == 0) {
106 float t_s = (float)s_tick / ACCEL_TICK_HZ; /* global time: phase-continuous across fs changes */
107 accel_sample_t s = {
108 .t_ms = s_tick * (1000 / ACCEL_TICK_HZ),
109 .axis = (axis_t)a,
110 .value = accel_sim_eval(cfg.func, cfg.amp_g, t_s),
111 };
112 if (xQueueSend(s_sample_queue, &s, 0) != pdTRUE) { /* 0 timeout: never block the timer */
113 s_dropped++; /* stream_task is stuck in uart_write_bytes(): the link is saturated */
114 }
115 }
116 }
117}

131static void flush_axis(axis_t axis)
132{
133 axis_batch_t *b = &s_batch[axis];
134 if (b->count == 0) {
135 return;
136 }
137 axis_cfg_t cfg = accel_sim_get_axis(axis);
138 char payload[LINK_MAX_FRAME];
139 int len = snprintf(payload, sizeof(payload), "ACC,%c,%lu,%u",
140 "XYZ"[axis], (unsigned long)b->t0_ms, (unsigned)cfg.fs_hz);
141 for (int i = 0; i < b->count && len > 0 && (size_t)len < sizeof(payload); i++) {
142 len += snprintf(payload + len, sizeof(payload) - (size_t)len, ",%.3f", (double)b->values[i]);
143 }
144 uart_link_send_frame(payload); /* logs + drops if still too long; see uart_link.c */
145 b->count = 0;
146}

161static void stream_task(void *arg)
162{
163 (void)arg;
164 accel_sample_t sample;
165 uint32_t reported_drops = 0;
166 int64_t last_report_us = 0;
167 for (;;) {
168 if (xQueueReceive(s_sample_queue, &sample, portMAX_DELAY) != pdTRUE) {
169 continue;
170 }
171
172 uint32_t dropped = s_dropped;
173 int64_t now_us = esp_timer_get_time();
174 if (dropped != reported_drops && now_us - last_report_us > 1000000) {
175 ESP_LOGW(TAG, "%lu samples dropped so far: link saturated, raise LINK_BAUD",
176 (unsigned long)dropped);
177 reported_drops = dropped;
178 last_report_us = now_us;
179 }
180
181 axis_batch_t *b = &s_batch[sample.axis];
182 if (b->count == 0) {
183 b->t0_ms = sample.t_ms;
184 }
185 if (b->count < ACCEL_BATCH_MAX) {
186 b->values[b->count++] = sample.value;
187 }
188
189 axis_cfg_t cfg = accel_sim_get_axis(sample.axis);
190 uint32_t target = (cfg.fs_hz * ACCEL_BATCH_MS) / 1000;
191 if (target < 1) {
192 target = 1;
193 }
194 if (b->count >= (int)target || b->count >= ACCEL_BATCH_MAX) {
195 flush_axis(sample.axis);
196 }
197 }
198}

217 xTaskCreate(stream_task, "accel_stream", 4096, NULL, 9, NULL);
218 ESP_LOGI(TAG, "init done (master tick %d Hz)", ACCEL_TICK_HZ);

241bool accel_sim_set_axis(axis_t axis, const axis_cfg_t *cfg)
242{
243 if (axis >= AXIS_COUNT || cfg == NULL || !accel_sim_valid_func(cfg->func) ||
244 !accel_sim_valid_amp(cfg->amp_g) || !accel_sim_valid_fs(cfg->fs_hz)) {
245 return false;
246 }
247 portENTER_CRITICAL(&s_cfg_lock);
248 s_cfg[axis] = *cfg;
249 portEXIT_CRITICAL(&s_cfg_lock);
250 ESP_LOGI(TAG, "axis %c: func=%d amp=%u g fs=%u Hz",
251 "XYZ"[axis], (int)cfg->func, (unsigned)cfg->amp_g, (unsigned)cfg->fs_hz);
252 return true;
253}
