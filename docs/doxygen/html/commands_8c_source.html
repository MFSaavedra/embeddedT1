85static void handle_cfg(char *fields[], int n)
86{
87 if (n != 5 || fields[1][0] == '\0' || fields[1][1] != '\0') {
88 send_err("BADARG");
89 return;
90 }
91 axis_t axis;
92 switch (fields[1][0]) {
93 case 'X': axis = AXIS_X; break;
94 case 'Y': axis = AXIS_Y; break;
95 case 'Z': axis = AXIS_Z; break;
96 default: send_err("BADARG"); return;
97 }
98
99 long func, amp, fs;
100 if (!parse_field(fields[2], 0, UINT8_MAX, &func) ||
101 !parse_field(fields[3], 0, UINT8_MAX, &amp) ||
102 !parse_field(fields[4], 0, UINT16_MAX, &fs)) {
103 send_err("BADARG");
104 return;
105 }
106 axis_cfg_t cfg = {
107 .func = (wave_func_t)func,
108 .amp_g = (uint8_t)amp,
109 .fs_hz = (uint16_t)fs,
110 };
111 if (!accel_sim_set_axis(axis, &cfg)) {
112 send_err("BADARG");
113 return;
114 }
115 send_ack("CFG");
116}

163static void handle_baud(char *fields[], int n)
164{
165 long baud;
166 if (n != 2 || !parse_field(fields[1], 0, 1000000, &baud) ||
167 !uart_link_valid_baud((unsigned)baud)) {
168 send_err("BADARG");
169 return;
170 }
171
172 accel_sim_stop();
173 env_sim_stop();
174
175 char payload[32];
176 snprintf(payload, sizeof payload, "ACK,BAUD,%ld", baud);
177 uart_link_send_frame(payload);
178
179 if (!uart_link_set_baud((unsigned)baud)) {
180 /* Still at the old rate: the PC will switch anyway on the ACK it has just been
181 * sent, stop hearing us, and fall back by itself. Nothing better to report. */
182 ESP_LOGE(TAG, "could not switch to %ld baud", baud);
183 }
184}

216void commands_handle_line(const char *line)
217{
218 char payload[LINK_MAX_LINE];
219 if (!proto_parse(line, payload, sizeof payload)) {
220 /* Noise, an echo, or a bad checksum: report it but never crash or reset state. */
221 ESP_LOGW(TAG, "rejected line: %s", line);
222 send_err("BADFRAME");
223 return;
224 }
225
226 char *fields[MAX_FIELDS];
227 int n = proto_split(payload, fields, MAX_FIELDS);
228 if (n == 0) {
229 send_err("EMPTY");
230 return;
231 }
232
233 /* A frame that passed the checksum proves the PC is talking at the current rate, so a
234 * link speed set by a recent BAUD command is now confirmed (no-op otherwise). */
235 uart_link_confirm_baud();
236
237 if (strcmp(fields[0], "CFG") == 0) {
238 handle_cfg(fields, n);
239 } else if (strcmp(fields[0], "ENV") == 0) {
240 handle_env(fields, n);
241 } else if (strcmp(fields[0], "INIT") == 0) {
242 handle_init(fields, n);
243 } else if (strcmp(fields[0], "BAUD") == 0) {
244 handle_baud(fields, n);
245 } else {
246 send_err("UNKNOWN");
247 }
248}
