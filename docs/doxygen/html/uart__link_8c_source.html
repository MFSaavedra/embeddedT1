64static void on_revert(void *arg)
65{
66 (void)arg;
67 s_revert_armed = false;
68 /* No ESP_ERROR_CHECK here: this is the recovery path. A timeout draining a TX buffer
69 * nobody is reading must not turn a bad baud rate into a panic. */
70 uart_wait_tx_done(LINK_UART_NUM, pdMS_TO_TICKS(200));
71 uart_set_baudrate(LINK_UART_NUM, LINK_BAUD);
72 uart_flush_input(LINK_UART_NUM);
73 ESP_LOGW(TAG, "no frame received at the new rate, reverted to %d baud", LINK_BAUD);
74}

76bool uart_link_set_baud(unsigned baud)
77{
78 if (!uart_link_valid_baud(baud)) {
79 return false;
80 }
81 if (s_revert_timer == NULL) {
82 const esp_timer_create_args_t args = {
83 .callback = on_revert,
84 .arg = NULL,
85 .dispatch_method = ESP_TIMER_TASK,
86 .name = "baud_revert",
87 };
88 ESP_ERROR_CHECK(esp_timer_create(&args, &s_revert_timer));
89 }
90 if (s_revert_armed) { /* a second switch before the first was confirmed */
91 esp_timer_stop(s_revert_timer);
92 s_revert_armed = false;
93 }
94
95 /* Everything already queued - the ACK that tells the PC to switch, and any log line
96 * ahead of it - must reach the wire at the OLD rate before the divisor changes. */
97 ESP_LOGI(TAG, "switching link to %u baud (provisional)", baud);
98 if (uart_wait_tx_done(LINK_UART_NUM, pdMS_TO_TICKS(200)) != ESP_OK ||
99 uart_set_baudrate(LINK_UART_NUM, baud) != ESP_OK) {
100 return false;
101 }
102 uart_flush_input(LINK_UART_NUM); /* bytes clocked in mid-switch are garbage */
103
104 s_revert_armed = true;
105 ESP_ERROR_CHECK(esp_timer_start_once(s_revert_timer, (uint64_t)LINK_BAUD_REVERT_MS * 1000ULL));
106 return true;
107}

109void uart_link_confirm_baud(void)
110{
111 if (!s_revert_armed) {
112 return;
113 }
114 s_revert_armed = false;
115 /* Unchecked on purpose: the revert timer may have fired between the test above and
116 * here (RX task vs. esp_timer task), and stopping an expired timer is not an error
117 * worth panicking over - the rate is back to LINK_BAUD either way. */
118 esp_timer_stop(s_revert_timer);
119 ESP_LOGI(TAG, "link speed confirmed by the PC");
120}

149static void rx_task(void *arg)
150{
151 (void)arg;
152 static char line[LINK_MAX_LINE];
153 size_t pos = 0;
154 uint8_t buf[64];
155
156 for (;;) {
157 int n = uart_read_bytes(LINK_UART_NUM, buf, sizeof buf, pdMS_TO_TICKS(20));
158 for (int i = 0; i < n; i++) {
159 char c = (char)buf[i];
160 if (c == '\n' || c == '\r') {
161 if (pos > 0) {
162 line[pos] = '\0';
163 s_line_cb(line);
164 pos = 0;
165 }
166 } else if (pos < sizeof line - 1) {
167 line[pos++] = c;
168 } else {
169 pos = 0; /* overlong garbage: discard the line */
170 }
171 }
172 }
173}
