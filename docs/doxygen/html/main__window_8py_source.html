123 """@brief Ask the worker to open @p port at @p baud (result arrives via signals).

135 def _on_connected(self, port: str, baud: int) -> None:
136 """@brief Port open and board rebooted: enable the controls and clear stale data.
137
138 The port was opened at the board's boot speed. If the user asked for a different one
139 - before connecting, or during an earlier session - the link is renegotiated now, so
140 the selector means the same thing whenever it is touched.
141
142 @param port Device name that was opened.
143 @param baud Baud rate the port was opened at.
144 """
145 self._baud = baud
146 self.connection.set_connected(True, f"Conectado a {port} @ {baud}")
147 self.config.setEnabled(True)
148 self.plots.clear()
149 self.env.clear()
150 target = self.connection.target_baud()
151 if target != baud:

200 # --------------------------- baud renegotiation --------------------------
201 # Both ends cannot switch at the same instant, so they switch in order: the firmware
202 # answers at the old speed and only then changes its divisor, and the PC changes its
203 # own when that answer arrives. Every step is driven from _on_frame():
204 #
205 # selector ──▶ _request_baud_change ──▶ $BAUD,<n> ─────────────────────────▶ ESP32
206 # $ACK,BAUD,<n> ──▶ worker.request_baud ──▶ baud_changed ──▶ $INIT ────────▶ ESP32
207 # $ACK,INIT ──▶ _finish_baud_change (also what tells the firmware to keep the rate)
208 #
209 # If any step is missed the firmware falls back after LINK_BAUD_REVERT_MS and
210 # _on_baud_timeout() brings the PC back to the same speed.
211
212 def _request_baud_change(self, baud: int) -> None:
213 """@brief Ask the firmware to move the link to @p baud.
214
215 The selector is locked until the outcome is known, so a second request cannot be
216 issued while the two ends are mid-switch.
217
218 @param baud Rate chosen in the connection panel; one of its BAUD_RATES.
219 """
220 if baud == self._baud:
221 return
222 self._baud_pending = baud
223 self.connection.set_busy(True)
224 self.connection.set_connected(True, f"Cambiando a {baud} baud…")
225 if not self.worker.send(f"{protocol.CMD_BAUD},{baud}"):
226 self._abort_baud_change("No se pudo enviar el cambio de velocidad")
227 return
228 self._baud_timer.start(BAUD_NEGOTIATION_MS)
229
230 def _on_baud_changed(self, _baud: int) -> None:
231 """@brief The port changed speed: say something at the new speed, namely INIT.
232
233 The firmware treats any valid frame as confirmation and keeps the speed it switched
234 to; INIT is the one to send because a BAUD command also stops the streaming, which
235 INIT restarts. This runs for the fallback switch of _abort_baud_change() too, where
236 it is equally what the link needs: the firmware has reverted and is waiting to be
237 started again.
238
239 @param _baud Speed the port is now running at (unused; the state is in _baud).
240 """
241 self.worker.send(protocol.CMD_INIT)
242
243 def _finish_baud_change(self) -> None:
244 """@brief A frame came back at the new speed: the renegotiation is complete."""
245 self._baud = self._baud_pending or self._baud
246 self._baud_pending = None
247 self._baud_timer.stop()
248 self.connection.set_busy(False)
249 self.connection.set_connected(True, f"Conectado a {self._port_name()} @ {self._baud}")
250
251 def _abort_baud_change(self, why: str) -> None:
252 """@brief Give up on the renegotiation and put both ends back on the old speed.
253
254 The firmware reverts on its own once no frame reaches it (LINK_BAUD_REVERT_MS), so
255 restoring the PC side is enough to make the link usable again. If the PC had already
256 switched, putting it back also triggers _on_baud_changed() and therefore an INIT,
257 which restarts the streaming the BAUD command stopped; if it never switched (a rate
258 the firmware refused outright) nothing was stopped in the first place.
259
260 @param why Reason shown in the status bar.
261 """
262 self._baud_timer.stop()
263 self._baud_pending = None
264 self.worker.request_baud(self._baud)
265 self.connection.set_baud(self._baud)
266 self.connection.set_busy(False)
267 self.connection.set_connected(True, f"Conectado a {self._port_name()} @ {self._baud}")
268 self.statusBar().showMessage(
269 f"{why}: se volvió a {self._baud} baud. "
270 "Si no llegan datos, pulsa «Inicializar ESP32».", 8000)
271
272 def _on_baud_timeout(self) -> None:
273 """@brief Nothing came back at the new speed within BAUD_NEGOTIATION_MS."""
274 self._abort_baud_change("El ESP32 no respondió al cambio de velocidad")

288 def _on_frame(self, fields: List[str]) -> None:
289 """@brief Route one valid frame by its type and update the counters.
290
291 - ACC: fields are axis, t0_ms, fs_hz, then the sample values -> AccelPlots.push().
292 - ENV: temperature and humidity -> EnvPanel.update_values().
293 - ACK: shown in the status bar; ACK,INIT also resets the config panel and plots to
294 the spec defaults, since the firmware has just done the same, and ACK,BAUD is the
295 cue to move the PC side of the link to the new speed (see the renegotiation
296 section above).
297 - ERR: shown in the status bar; it also aborts a renegotiation in progress, which
298 is how a rate the firmware refuses outright is handled without waiting.
299
300 A frame with a valid checksum but malformed fields is counted as rejected instead of
301 valid, so the two counters stay mutually exclusive.
302
303 @param fields Payload fields from SerialWorker.frame_received; fields[0] is the type.
304 """
305 self._frames_ok += 1
306 kind = fields[0]
307 if kind == protocol.MSG_ACC:
308 try:
309 axis, t0_ms, fs_hz = fields[1], int(fields[2]), int(fields[3])
310 values = [float(v) for v in fields[4:]]
311 self.plots.push(axis, t0_ms, fs_hz, values)
312 except (ValueError, IndexError):
313 self._frames_ok -= 1
314 self._frames_bad += 1
315 elif kind == protocol.MSG_ENV:
316 try:
317 temp_c, hum_pct = float(fields[1]), int(fields[2])
318 self.env.update_values(temp_c, hum_pct)
319 except (ValueError, IndexError):
320 self._frames_ok -= 1
321 self._frames_bad += 1
322 elif kind == protocol.MSG_ACK:
323 self.statusBar().showMessage("ESP32: OK " + ",".join(fields[1:]), 3000)
324 if len(fields) > 1 and fields[1] == protocol.CMD_INIT:
325 self.config.reset_defaults()
326 self.plots.clear()
327 self.env.clear() # the firmware sends a fresh $ENV right after this ACK
328 for axis in self.config.axes:
329 self.plots.set_amplitude(axis, 4)
330 if self._baud_pending is not None:
331 self._finish_baud_change() # a frame at the new speed: it works
332 elif len(fields) > 1 and fields[1] == protocol.CMD_BAUD:
333 try:
334 self.worker.request_baud(int(fields[2]))
335 except (ValueError, IndexError):
336 self._frames_ok -= 1
337 self._frames_bad += 1
338 self._abort_baud_change("Respuesta de velocidad ilegible")
339 elif kind == protocol.MSG_ERR:
340 self.statusBar().showMessage("ESP32: ERROR " + ",".join(fields[1:]), 5000)
341 if self._baud_pending is not None:
342 self._abort_baud_change("El ESP32 rechazó la velocidad")
343 self._update_counters()
