36def _combo(items: Sequence[Tuple[int, str]], default: int) -> QComboBox:
37 """@brief Build a QComboBox from (value, label) pairs with @p default preselected.
38
39 @param items Sequence of (item data, display text).
40 @param default Item data to select initially; must be present in @p items.
41 @return The populated combo box; currentData() yields the integer value.
42 """
43 combo = QComboBox()
44 for value, label in items:
45 combo.addItem(label, value)
46 combo.setCurrentIndex([v for v, _ in items].index(default))
47 return combo
