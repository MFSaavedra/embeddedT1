22class ConnectionPanel(QGroupBox):
23 """@brief Port / baud selectors plus the Conectar and Inicializar ESP32 buttons.
24
25 The panel only emits requests; MainWindow performs them and reports back through
26 set_ports() and set_connected().
27
28 The baud selector means one thing in both states: the speed the *link* should run at.
29 It is never the speed the port is opened at - that is always DEFAULT_BAUD, because the
30 board boots into its compile-time rate. Picking another rate while connected
31 renegotiates (baud_change_requested); picking one while disconnected is remembered and

101 def set_ports(self, ports: List[Tuple[str, str]]) -> None:
102 """@brief Repopulate the port list, keeping the current choice if it still exists.
103
104 The combo shows the label; the device name travels as item data so that
105 connect_requested carries exactly what pyserial must open. The first entry is
106 preselected, which is the first USB adapter when available_ports() found one.
107
108 @param ports (device, label) pairs in display order, from
109 serial_worker.available_ports().
110 """
111 current = self.port_combo.currentData()
112 self.port_combo.clear()
113 for device, label in ports:
114 self.port_combo.addItem(label, device)
115 index = self.port_combo.findData(current)
116 if index >= 0:
117 self.port_combo.setCurrentIndex(index)
118 if not ports:
119 self.status_label.setText("No se encontraron puertos")

121 def set_connected(self, connected: bool, text: Optional[str] = None) -> None:
122 """@brief Reflect the connection state: button labels, enabled widgets, status text.
123
124 While connected the port and refresh controls are locked and Inicializar is enabled;
125 while disconnected the opposite. What set_busy() locks stays locked either way.
126
127 @param connected True if a port is open.
128 @param text Status text; defaults to "Conectado" / "Desconectado".
129 """
130 self._connected = connected
131 self._apply_enabled_state()
132 self.status_label.setText(text or ("Conectado" if connected else "Desconectado"))

134 def set_busy(self, busy: bool) -> None:
135 """@brief Lock the controls that must not be used mid-renegotiation.
136
137 Both the baud selector (a second switch while one is in flight) and Inicializar (an
138 INIT written at the old speed while the firmware has already left it) would land in
139 the middle of the changeover.
140
141 @param busy True between the BAUD command and its outcome.
142 """
143 self._busy = busy
144 self._apply_enabled_state()

146 def _apply_enabled_state(self) -> None:
147 """@brief Single place that decides which controls are usable.
148
149 Both the connection state and the busy flag feed into it, so neither can silently
150 undo the other - set_connected() used to re-enable what a renegotiation had locked.
151 """
152 self.connect_btn.setText("Desconectar" if self._connected else "Conectar")
153 self.connect_btn.setEnabled(not self._busy)
154 self.init_btn.setEnabled(self._connected and not self._busy)
155 self.baud_combo.setEnabled(not self._busy)
156 for w in (self.port_combo, self.refresh_btn):
157 w.setEnabled(not self._connected and not self._busy)

169 def set_baud(self, baud: int) -> None:
170 """@brief Show @p baud on the selector without emitting baud_change_requested.
171
172 Used when the link speed changes for a reason other than the user picking it, i.e.
173 when a renegotiation fails and the GUI has to fall back.
174
175 @param baud Rate to display; ignored if it is not one of BAUD_RATES.
176 """
177 index = self.baud_combo.findData(baud)
178 if index < 0:
179 return
180 self.baud_combo.blockSignals(True)
181 self.baud_combo.setCurrentIndex(index)
182 self.baud_combo.blockSignals(False)

184 def _on_baud_changed(self, _index: int) -> None:
185 """@brief Slot for the baud selector: renegotiate now, or leave it to the connect.
186
187 While connected this is a request to change the speed of a running link, which only
188 the firmware can grant. While disconnected there is nothing to renegotiate yet, so
189 the value is simply left for target_baud() to be read at connect time.
190
191 @param _index New combo index (unused; currentData() is read instead).
192 """
193 if self._connected:
194 self.baud_change_requested.emit(self.baud_combo.currentData())

196 def _on_connect_clicked(self) -> None:
197 """@brief Emit disconnect_requested or connect_requested depending on the state.
198
199 The requested speed is always DEFAULT_BAUD; see _on_baud_changed() for what the baud
200 selector does instead. Does nothing (beyond a hint in the status label) if no port is
201 selected.
202 """
203 if self._connected:
204 self.disconnect_requested.emit()
205 return
206 port = self.port_combo.currentData()
207 if not port:
208 self.status_label.setText("Selecciona un puerto")
209 return
210 # Always DEFAULT_BAUD, never the selector: the board has just been reset and boots
211 # into its compile-time rate. A different selection is honoured by renegotiating
212 # once the link is up (MainWindow._on_connected -> target_baud()).
213 self.connect_requested.emit(port, DEFAULT_BAUD)
