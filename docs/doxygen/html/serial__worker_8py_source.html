1"""@file serial_worker.py
2@brief Background serial I/O so the GUI thread never blocks (spec 7: "usar threading").
3
4SerialWorker owns the pyserial port. Its read loop runs in a QThread and talks to the GUI
5only through Qt signals (thread-safe, delivered on the GUI thread). Writes are serialised
6with a lock so any widget may call send() directly from the GUI thread.
7
8Anything that reconfigures the port instead of writing to it - request_baud() - is queued
9and applied by the read loop between two reads, because the loop is normally blocked inside
10readline() and reconfiguring underneath it corrupts that read.

35def available_ports() -> List[Tuple[str, str]]:
36 """@brief Serial ports currently present, USB-serial adapters first.
37
38 Legacy motherboard ports (/dev/ttyS0, COM1) open without error and then just swallow
39 everything, so they are listed after any USB adapter: with the ESP32 plugged in, its
40 CP2102 is the preselected entry in the connection panel and its label says so.
41
42 @return (device, label) pairs such as
43 ("/dev/ttyUSB0", "/dev/ttyUSB0 — Silicon Labs CP2102 USB to UART Bridge
44 Controller") or ("/dev/ttyS0", "/dev/ttyS0"); empty if no port is present.
45 """
46 ports = list_ports.comports()
47 usb = sorted((p for p in ports if p.vid is not None), key=lambda p: p.device)
48 other = sorted((p for p in ports if p.vid is None), key=lambda p: p.device)
49 return [(p.device, _port_label(p)) for p in usb + other]

52def _port_label(info) -> str:
53 """@brief Human-readable combo entry for one port: "<device> — <manufacturer product>".
54
55 Falls back to pyserial's description when the USB product string is missing (Windows)
56 and to the bare device name when even that is uninformative (pyserial reports "ttyS0"
57 or "n/a" for legacy ports).
58
59 @param info A serial.tools.list_ports_common.ListPortInfo.
60 @return The label; the device name alone if nothing better is known.
61 """
62 what = info.product or info.description or ""
63 if what in ("n/a", os.path.basename(info.device)):
64 what = ""
65 if info.manufacturer and info.manufacturer not in what:
66 what = f"{info.manufacturer} {what}".strip()
67 return f"{info.device} — {what}" if what else info.device

70class SerialWorker(QThread):
71 """@brief Owns the serial port and runs the read loop in its own thread.
72
73 Lifecycle: open() starts run() in the background, which first resets the board (see
74 _reset_board()), waits BOOT_WAIT_S for it to boot and emits connected, then emits
75 frame_received / bad_line for every line until close() is called or the port fails.
76 send() and request_baud() may be called from any thread while the port is open.
77 """

112 def open(self, port: str, baud: int) -> bool:
113 """@brief Open the port and start the read loop.
114
115 Returns as soon as the port is open; the board reset and boot wait happen in run(),
116 so connected arrives roughly BOOT_WAIT_S later via the signal, not before this
117 returns.
118
119 @param port Device name, e.g. "/dev/ttyUSB0" or "COM3".
120 @param baud Baud rate; must match LINK_BAUD in the firmware.
121 @return True if the port was opened (connected will be emitted); False if the
122 worker is already running or the port could not be opened (error is
123 emitted).
124 """
125 if self.isRunning():
126 return False
127 try:
128 self._ser = serial.Serial(port, baudrate=baud, timeout=0.1)
129 except (serial.SerialException, OSError, ValueError) as exc:
130 self.error.emit(f"No se pudo abrir {port}: {exc}")
131 return False
132 self._stop.clear()
133 self.start()
134 return True

136 def close(self) -> None:
137 """@brief Stop the read loop and close the port.
138
139 Blocks up to 2 s for the thread to finish, then closes the port whether or not it
140 did: readline() can sit on a byte stream that never yields a newline (the board
141 talking at another speed), so the wait is bounded and the reader is written to
142 tolerate having the port closed under it (see _pump_line()). Safe to call when not
143 connected.
144 """
145 self._stop.set()
146 if self.isRunning():
147 self.wait(2000)
148 self._close_port()

150 def request_baud(self, baud: int) -> None:
151 """@brief Ask the read loop to switch the open port to @p baud.
152
153 The switch is deferred instead of applied here: run() is normally blocked inside
154 readline(), and reconfiguring the port under it would corrupt that read. The loop
155 picks the request up within one read timeout (0.1 s) and emits baud_changed once
156 the port is running at the new speed.
157
158 Only the PC side changes. The firmware must have been told separately (BAUD
159 command) and must already have switched, which is what its ACK means.
160
161 @param baud New speed; must be one the firmware accepts.
162 """

164 self._pending_baud = baud

166 def send(self, payload: str) -> bool:
167 """@brief Frame and transmit a command payload.
168
169 @param payload Payload text such as "CFG,X,1,4,100" (see protocol.py).
170 @return True if the frame was written; False if not connected or the write failed
171 (error is emitted in both cases).
172 """
173 ser = self._ser
174 if ser is None or not ser.is_open:
175 self.error.emit("No conectado")
176 return False
177 try:
178 with self._write_lock:
179 ser.write(protocol.build_frame(payload))
180 except (serial.SerialException, OSError) as exc:
181 self.error.emit(f"Error al enviar: {exc}")
182 return False
183 return True

187 def run(self) -> None:
188 """@brief Thread body: reset the board, wait for it to boot, then read lines until
189 stopped or the port fails.
190
191 Every line is passed through protocol.parse_frame(): valid frames are emitted as
192 frame_received, everything else as bad_line. Lines received during the boot wait
193 (ROM output at 115200, bootloader and app logs) take the same path, so they show
194 up in the rejected counter like any other log line. On an I/O error (typically an
195 unplugged cable) error is emitted and the loop ends. The port is always closed and
196 disconnected emitted on exit.
197 """
198 ser = self._ser
199 try:
200 self._run_until_stopped(ser)
201 except Exception as exc: # noqa: BLE001 - the contract is that run() never raises
202 self.error.emit(f"Error interno del hilo serial: {exc}")
203 finally:
204 self._close_port()
205 self.disconnected.emit()

207 def _run_until_stopped(self, ser: serial.Serial) -> None:
208 """@brief Body of run(): reset, wait for boot, then read until stopped.
209
210 Split out so run() can be a single try/finally that lets nothing escape.
211
212 @param ser The open port.
213 """
214 self._reset_board(ser)
215 boot_deadline = time.monotonic() + BOOT_WAIT_S
216 while not self._stop.is_set() and time.monotonic() < boot_deadline:
217 if not self._pump_line(ser):
218 return
219 if self._stop.is_set():
220 return
221 self.connected.emit(ser.port, ser.baudrate)
222 while not self._stop.is_set():
223 if not self._pump_line(ser):
224 return
225 self._apply_pending_baud(ser)

227 def _reset_board(self, ser: serial.Serial) -> None:
228 """@brief Pulse EN low through the dev board's DTR/RTS auto-reset circuit.
229
230 Same sequence as esptool's hard reset: DTR deasserted keeps IO0 high so the chip
231 boots the firmware (not the ROM download mode) while RTS is asserted for
232 RESET_PULSE_S to pull EN low. Merely opening the port also toggles both lines,
233 but the OS drives them in an order and with a timing that leave the ESP32 hung
234 most of the time on Linux with a CP2102 board (measured 1-4 usable connects out
235 of 10); an explicit pulse boots it every time. Both lines are left deasserted (EN
236 and IO0 high). A port without modem lines is reported but not fatal: the user can
237 still press the EN button.
238
239 @param ser The open port.
240 """
241 try:
242 ser.dtr = False
243 ser.rts = True
244 time.sleep(RESET_PULSE_S)
245 ser.rts = False
246 except (serial.SerialException, OSError) as exc:
247 self.error.emit(f"No se pudo reiniciar el ESP32 (pulsa EN en la placa): {exc}")
248
249 def _pump_line(self, ser: serial.Serial) -> bool:
250 """@brief Read one line (or time out) and emit frame_received / bad_line for it.
251
252 @param ser The open port.
253 @return False if the port failed (error has been emitted), True otherwise.
254 """
255 try:
256 raw = ser.readline() # b"" on timeout
257 except Exception as exc: # noqa: BLE001 - see below
258 # Typical cause: the cable was unplugged. TODO(PY1): reconnect policy.
259 # Deliberately broad: a port that dies mid-read does not only raise
260 # SerialException. Observed on a real CP2102 when close() closed the port
261 # while this thread was inside readline(): pyserial then calls os.read() with
262 # a None file descriptor and raises TypeError, which used to escape run().
263 self.error.emit(f"Conexión perdida: {exc}")
264 return False
265 if raw:
266 line = raw.decode("ascii", errors="replace")
267 fields = protocol.parse_frame(line)
268 if fields is None:
269 self.bad_line.emit(line.rstrip())
270 else:
271 self.frame_received.emit(fields)
272 return True

274 def _apply_pending_baud(self, ser: serial.Serial) -> None:
275 """@brief Apply a request_baud() if one is pending (called between reads).
276
277 Waits BAUD_SETTLE_S first so the board can finish sending, at the old speed, the log
278 line that trails the ACK, then drops those bytes: whatever is in the input buffer at
279 this point was clocked in at a speed that no longer applies.
280
281 A failure is reported but is not fatal: the port keeps its previous speed, which is
282 also the speed the firmware falls back to when no frame reaches it (LINK_BAUD_REVERT_MS),
283 so the link repairs itself instead of dying.
284
285 @param ser The open port.
286 """
287 with self._baud_lock:
288 baud, self._pending_baud = self._pending_baud, None
289 if baud is None or baud == ser.baudrate:
290 return
291 time.sleep(BAUD_SETTLE_S)
292 try:
293 ser.reset_input_buffer()
294 ser.baudrate = baud
295 except (serial.SerialException, OSError, ValueError) as exc:
296 self.error.emit(f"No se pudo cambiar a {baud} baud: {exc}")
297 return
298 self.baud_changed.emit(baud)

301 """@brief Close and forget the port, swallowing any error from a dead device."""
302 ser, self._ser = self._ser, None
303 if ser is not None:
304 try:
305 ser.close()
306 except Exception: # noqa: BLE001 - closing a dead port must never raise
307 pass
