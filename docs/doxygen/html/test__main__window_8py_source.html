66def test_successful_renegotiation(window):
67 """@brief Selector -> BAUD -> ACK,BAUD -> switch -> INIT -> ACK,INIT completes the change."""
68 _pick_baud(window, 230400)
69 assert window.worker.sent == ["BAUD,230400"]
70 assert window._baud_pending == 230400
71 # Locked while in flight: a second switch, or an INIT written at the speed the firmware
72 # has already left, would land in the middle of the changeover.
73 assert not window.connection.baud_combo.isEnabled()
74 assert not window.connection.init_btn.isEnabled()
75
76 # The firmware answers at the old speed, then switches; the PC follows.
77 window._on_frame(["ACK", "BAUD", "230400"])
78 assert window.worker.baud_requests == [230400]
79
80 # The worker reports the port is at the new speed: INIT proves it to the firmware.
81 window._on_baud_changed(230400)
82 assert window.worker.sent == ["BAUD,230400", "INIT"]
83 assert window._baud_pending == 230400 # not committed yet
84
85 window._on_frame(["ACK", "INIT", "1.0"])
86 assert window._baud == 230400
87 assert window._baud_pending is None
88 assert window.connection.baud_combo.isEnabled()
89 assert window.connection.init_btn.isEnabled()
90 assert not window._baud_timer.isActive()
