81def test_connect_pulses_en_then_waits_for_boot_before_connected(monkeypatch):
82 monkeypatch.setattr(serial, "Serial", FakePort)
83 monkeypatch.setattr(serial_worker, "RESET_PULSE_S", 0.01)
84 monkeypatch.setattr(serial_worker, "BOOT_WAIT_S", 0.3)
85 FakePort.instances.clear()
86
87 events = []
88 connected = threading.Event()
89 worker = SerialWorker()
90 direct = Qt.DirectConnection
91 worker.bad_line.connect(lambda line: events.append(("bad", line)), direct)
92 worker.frame_received.connect(lambda fields: events.append(("frame", fields)), direct)
93 worker.connected.connect(
94 lambda port, baud: (events.append(("connected", port, baud)), connected.set()), direct)
95 worker.error.connect(lambda msg: events.append(("error", msg)), direct)
96
97 t0 = time.monotonic()
98 assert worker.open("FAKE", 921600)
99 assert connected.wait(2.0), events
100 elapsed = time.monotonic() - t0
101 worker.close()
102
103 port = FakePort.instances[0]
104 # esptool-style hard reset: IO0 high (DTR off), EN low (RTS on), EN released.
105 assert port.modem_events == [("dtr", False), ("rts", True), ("rts", False)]
106 assert not port.is_open
107
108 kinds = [e[0] for e in events]
109 assert "error" not in kinds
110 # connected comes after the boot wait, and boot output was still reported as bad lines.
111 assert elapsed >= 0.3
112 assert kinds.index("connected") > kinds.index("bad")

118def test_request_baud_is_applied_by_the_read_loop_not_the_caller(monkeypatch):
119 """@brief request_baud() defers the switch to the reader thread.
120
121 Reconfiguring the port from the GUI thread would corrupt the readline() that the worker
122 is blocked in, so the switch must happen in run(), between reads, after the input
123 buffered at the old speed has been dropped.
124 """
125 monkeypatch.setattr(serial, "Serial", FakePort)
126 monkeypatch.setattr(serial_worker, "RESET_PULSE_S", 0.01)
127 monkeypatch.setattr(serial_worker, "BOOT_WAIT_S", 0.1)
128 monkeypatch.setattr(serial_worker, "BAUD_SETTLE_S", 0.01)
129 FakePort.instances.clear()
130
131 connected = threading.Event()
132 switched = threading.Event()
133 seen = []
134 worker = SerialWorker()
135 direct = Qt.DirectConnection
136 worker.connected.connect(lambda *_: connected.set(), direct)
137 worker.baud_changed.connect(lambda b: (seen.append(b), switched.set()), direct)
138 worker.error.connect(lambda msg: seen.append(("error", msg)), direct)
139
140 assert worker.open("FAKE", 921600)
141 assert connected.wait(2.0)
142 port = FakePort.instances[0]
143 assert port.baud_events == [] # opening the port must not reconfigure it
144
145 worker.request_baud(230400)
146 assert switched.wait(2.0), seen
147 worker.close()
148
149 assert seen == [230400]
150 # Input flushed first, then the divisor changed - both from the worker thread.
151 assert [what for what, _ in port.baud_events] == ["reset", 230400]
152 assert all(ident != threading.get_ident() for _, ident in port.baud_events)
153 assert port.baudrate == 230400
